package mx.syncro.chispazo;
import android.app.*; import android.os.*; import android.graphics.Color; import android.graphics.Typeface; import android.view.*; import android.widget.*; import java.util.*;
public class MainActivity extends Activity {
    Db db; LinearLayout body; TextView status; List<Draw> draws=new ArrayList<>(); Analyzer an;
    int purple=Color.rgb(106,27,154), ink=Color.rgb(28,31,38), muted=Color.rgb(105,112,125), bg=Color.rgb(245,247,251);
    public void onCreate(Bundle b){super.onCreate(b);db=new Db(this); build(); loadLocal(); if(draws.isEmpty()) refresh();}
    TextView tv(String s,int sp,boolean bold){TextView v=new TextView(this);v.setText(s);v.setTextSize(sp);v.setTextColor(ink);v.setPadding(8,8,8,8);if(bold)v.setTypeface(Typeface.DEFAULT,Typeface.BOLD);return v;}
    Button btn(String s){Button b=new Button(this);b.setText(s);b.setAllCaps(false);return b;}
    void build(){LinearLayout root=new LinearLayout(this);root.setOrientation(LinearLayout.VERTICAL);root.setPadding(24,20,24,12);root.setBackgroundColor(bg);TextView title=tv("Chispazo Analítica",26,true);title.setTextColor(purple);root.addView(title);status=tv("Cargando histórico…",13,false);status.setTextColor(muted);root.addView(status);
        HorizontalScrollView hs=new HorizontalScrollView(this);LinearLayout nav=new LinearLayout(this);String[] labs={"Resumen","Frecuencias","Pares","Tríos","Sugerencias","Historial"};for(String l:labs){Button b=btn(l);b.setOnClickListener(v->show(l));nav.addView(b);}hs.addView(nav);root.addView(hs);
        Button upd=btn("Actualizar desde Lotería Nacional");upd.setOnClickListener(v->refresh());root.addView(upd);
        ScrollView sv=new ScrollView(this);body=new LinearLayout(this);body.setOrientation(LinearLayout.VERTICAL);body.setPadding(0,10,0,60);sv.addView(body);root.addView(sv,new LinearLayout.LayoutParams(-1,0,1));setContentView(root);}
    void loadLocal(){draws=db.all();if(!draws.isEmpty()){an=new Analyzer(draws);status.setText(draws.size()+" sorteos guardados · último #"+draws.get(0).contest+" · "+draws.get(0).date);show("Resumen");}}
    void refresh(){status.setText("Descargando CSV oficial…"); new Thread(()->{try{List<Draw>d=CsvClient.fetch();if(d.isEmpty())throw new Exception("CSV sin sorteos reconocibles");db.replaceAll(d);runOnUiThread(()->{loadLocal();Toast.makeText(this,"Histórico actualizado",Toast.LENGTH_SHORT).show();});}catch(Exception e){runOnUiThread(()->{status.setText("No se pudo actualizar: "+e.getMessage()+" · se mantienen los datos locales");if(!draws.isEmpty())show("Resumen");});}}).start();}
    void clear(){body.removeAllViews();}
    void show(String tab){if(an==null)return;clear(); switch(tab){case"Frecuencias":freq();break;case"Pares":pairs();break;case"Tríos":trios();break;case"Sugerencias":picks();break;case"Historial":history();break;default:summary();}}
    void heading(String s){TextView x=tv(s,20,true);x.setTextColor(purple);body.addView(x);}void line(String s){body.addView(tv(s,16,false));}
    void summary(){heading("Panorama");Draw d=draws.get(0);line("Último sorteo disponible: #"+d.contest+" · "+d.date);line("Ganadora: "+d.numbers());line("Sorteos analizados: "+draws.size());line(String.format(Locale.US,"Suma histórica media: %.1f",an.meanSum()));heading("Más fuertes por frecuencia ponderada");int i=0;for(Analyzer.Stat s:an.numbers()){line((++i)+". "+s.key+" · apariciones "+s.count+" · índice "+String.format(Locale.US,"%.2f",s.score));if(i==10)break;}heading("Nota");line("Las estadísticas describen el histórico. Cada combinación válida sigue teniendo la misma probabilidad teórica en un sorteo independiente.");}
    void freq(){heading("Frecuencia ponderada");int i=0;for(Analyzer.Stat s:an.numbers())line(String.format(Locale.US,"%02d. Número %s · total %d · índice %.2f",++i,s.key,s.count,s.score));}
    void pairs(){heading("Pares que más han coincidido");int i=0;for(Analyzer.Stat s:an.topPairs()){line((++i)+". "+s.key.replace("-"," + ")+" · "+s.count+" veces");if(i==40)break;}}
    void trios(){heading("Tríos que más han coincidido");int i=0;for(Analyzer.Stat s:an.topTrios()){line((++i)+". "+s.key.replace("-"," + ")+" · "+s.count+" veces");if(i==40)break;}}
    void picks(){heading("Combinaciones sugeridas");line("Ranking heurístico; no aumenta la probabilidad matemática del sorteo.");int i=0;for(Analyzer.Pick p:an.generate(80)){TextView a=tv((++i)+".  "+p.nums(),21,true);a.setTextColor(purple);body.addView(a);line(String.format(Locale.US,"Puntaje %.2f · %s",p.score,p.why));if(i==10)break;}}
    void history(){heading("Historial local");EditText q=new EditText(this);q.setHint("Buscar sorteo, fecha o número");body.addView(q);LinearLayout list=new LinearLayout(this);list.setOrientation(LinearLayout.VERTICAL);body.addView(list);Runnable render=()->{list.removeAllViews();String z=q.getText().toString().trim();int c=0;for(Draw d:draws){String row="#"+d.contest+" · "+d.date+" · "+d.numbers();if(z.isEmpty()||row.contains(z)){list.addView(tv(row,15,false));if(++c>=250)break;}}};q.addTextChangedListener(new android.text.TextWatcher(){public void beforeTextChanged(CharSequence s,int st,int c,int a){}public void onTextChanged(CharSequence s,int st,int b,int c){render.run();}public void afterTextChanged(android.text.Editable e){}});render.run();}
}
